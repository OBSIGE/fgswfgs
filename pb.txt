-- Безопасный клик по кнопкам игры
local function PressButton(button)
    if not button then return false end
    if firesignal then
        pcall(function() firesignal(button.MouseButton1Click) end)
        pcall(function() firesignal(button.Activated) end)
        return true
    end
    pcall(function() button:Activate() end)
    return true
end

-- Поиск машины
local function GetVehicle()
    local char = LP.Character
    if not char then return nil end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if humanoid and humanoid.SeatPart and humanoid.SeatPart:IsA("VehicleSeat") then
        return humanoid.SeatPart.Parent
    end
    if workspace:FindFirstChild("Vehicles") then
        local myCar = workspace.Vehicles:FindFirstChild(LP.Name)
        if myCar then return myCar end
    end
    return nil
end

-- Линейный поиск квестовых ProximityPrompt в указанной папке
local function FindPromptInRoot(rootFolder)
    local char = LP.Character
    if not char then return nil end
    
    for _, prompt in pairs(rootFolder:GetDescendants()) do
        if prompt:IsA("ProximityPrompt") and prompt.Enabled then
            local parentPart = prompt.Parent
            if parentPart and parentPart:IsA("BasePart") and parentPart.Position.Y > -10 then
                local text = (prompt.ObjectText .. prompt.ActionText):lower()
                if text:find("job") or text:find("load") or text:find("unload") or text:find("deliver") or text:find("drop") or text:find("collect") then
                    return parentPart
                end
            end
        end
    end
    return nil
end

-- Главная функция поиска active маркера
local function GetActiveObjectiveTarget()
    local char = LP.Character
    if not char then return nil end

    -- Способ 1: Ищем квестовые указатели (Beam / Attachment)
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("Beam") or v:IsA("Attachment") then
            local p = v.Parent
            if p and p:IsA("BasePart") and p.Position.Y > -20 then
                if not p:IsDescendantOf(char) and (not GetVehicle() or not p:IsDescendantOf(GetVehicle())) then
                    return p
                end
            end
        end
    end

    -- Способ 2: Ищем кнопки взаимодействия в Workspace
    local promptTargetWorkspace = FindPromptInRoot(workspace)
    if promptTargetWorkspace then return promptTargetWorkspace end

    -- Способ 3: Ищем кнопки взаимодействия в локальной Камере
    local promptTargetCamera = FindPromptInRoot(workspace.CurrentCamera)
    if promptTargetCamera then return promptTargetCamera end

    -- Способ 4: Проверяем 3D иконки (BillboardGui)
    for _, bGui in pairs(workspace:GetDescendants()) do
        if bGui:IsA("BillboardGui") and bGui.Enabled then
            local p = bGui.Parent
            if p and p:IsA("BasePart") and p.Position.Y > -20 and p.Transparency < 1 then
                if not p:IsDescendantOf(char) then
                    return p
                end
            end
        end
    end

    return nil
end

-- Плавный полет
local function FlyToTarget(targetPart)
    local vehicle = GetVehicle()
    if not vehicle or not targetPart then return false end
    
    local primary = vehicle.PrimaryPart or vehicle:FindFirstChild("VehicleSeat") or vehicle:FindFirstChildWhichIsA("BasePart")
    if not primary then return false end

    for _, part in pairs(vehicle:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = false end
    end

    local flyPosition = Vector3.new(targetPart.Position.X, targetPart.Position.Y + CONFIG.FlyHeight, targetPart.Position.Z)

    -- Полет по горизонтали
    while CONFIG.Enabled and (Vector3.new(primary.Position.X, flyPosition.Y, primary.Position.Z) - flyPosition).Magnitude > 6 do
        if not GetVehicle() then return false end
        local currentPos = primary.Position
        local direction = (flyPosition - currentPos).Unit
        local step = direction * (CONFIG.TeleportSpeed * RunService.Heartbeat:Wait())
        
        if vehicle.PivotTo then
            vehicle:PivotTo(CFrame.new(currentPos + step, flyPosition))
        else
            vehicle:SetPrimaryPartCFrame(CFrame.new(currentPos + step, flyPosition))
        end
    end

    -- Вертикальный спуск
    Status.Text = "Статус: 🛬 Аккуратный спуск..."
    while CONFIG.Enabled and (primary.Position - targetPart.Position).Magnitude > 1.5 do
        if not GetVehicle() then return false end
        local currentCFrame = primary.CFrame
        local downStep = currentCFrame.Position + Vector3.new(0, -22 * RunService.Heartbeat:Wait(), 0)
        
        if downStep.Y < targetPart.Position.Y then 
            downStep = Vector3.new(downStep.X, targetPart.Position.Y, downStep.Z)
        end
        
        if vehicle.PivotTo then
            vehicle:PivotTo(CFrame.new(downStep, targetPart.Position + Vector3.new(0,0,1)))
        else
            vehicle:SetPrimaryPartCFrame(CFrame.new(downStep, targetPart.Position + Vector3.new(0,0,1)))
        end
        
        if (primary.Position.Y - targetPart.Position.Y) <= 0.8 then break end
    end

    if CONFIG.Enabled then
        if vehicle.PivotTo then vehicle:PivotTo(CFrame.new(targetPart.Position)) else vehicle:SetPrimaryPartCFrame(CFrame.new(targetPart.Position)) end
    end
    return true
end

-- Линейная прокрутка Prompts во время нахождения на точке
local function ProcessPromptsInRoot(rootFolder)
    local char = LP.Character
    if not char or not char.PrimaryPart then return end
    
    for _, prompt in pairs(rootFolder:GetDescendants()) do
        if prompt:IsA("ProximityPrompt") and prompt.Enabled then
            local dist = (char.PrimaryPart.Position - prompt.Parent.Position).Magnitude
            if dist < prompt.MaxActivationDistance + 8 then
                prompt:InputHoldBegin()
                task.wait(prompt.HoldDuration + 0.05)
                prompt:InputHoldEnd()
            end
        end
    end
end

-- Фоновый поток для UI и зажима кнопок "E"
local function BackgroundLoop()
    while true do
        task.wait(0.2)
        if CONFIG.Enabled then
            -- Сбор UI меню квеста
            pcall(function()
                local hud = LP.PlayerGui:FindFirstChild("MainHUD") or LP.PlayerGui:FindFirstChild("HUD")
                if hud then
                    for _, v in pairs(hud:GetDescendants()) do
                        if v:IsA("GuiButton") and (v.Name:lower():find("accept") or v.Name:lower():find("claim") or v.Name:lower():find("take")) then
                            if v.Visible or (v.Parent and v.Parent.Visible) then
                                PressButton(v)
                            end
                        end
                    end
                end
            end)

            -- Взаимодействие с ProximityPrompts
            pcall(function() ProcessPromptsInRoot(workspace) end)
            pcall(function() ProcessPromptsInRoot(workspace.CurrentCamera) end)
        end
    end
end
task.spawn(BackgroundLoop)
